```python
import json, re, sys, urllib.parse, urllib.request, pathlib, time

QUESTS = [
  "The New Frontier Quest", "Children of the Revolution Quest", "Wrath of the Emperor Quest",
  "In Service of Yalahar Quest", "The Inquisition Quest", "The Pits of Inferno Quest",
  "Ferumbras' Ascendant Quest", "The Ice Islands Quest", "Threatened Dreams Quest",
  "The Dream Courts Quest", "The Secret Library Quest", "Forgotten Knowledge Quest",
  "Feaster of Souls Quest", "Soul War Quest", "Barbarian Test Quest",
]
UA = "tibia-quest-graph/0.1 (fansite data curation; gabrielfernandoi@gmail.com)"
OUT = pathlib.Path(__file__).parent / "pages"
OUT.mkdir(exist_ok=True)

def fetch(title):
    url = "https://tibia.fandom.com/api.php?" + urllib.parse.urlencode(
        {"action": "parse", "page": title, "prop": "wikitext", "format": "json", "formatversion": 2})
    req = urllib.request.Request(url, headers={"User-Agent": UA})
    with urllib.request.urlopen(req, timeout=30) as r:
        d = json.load(r)
    if "error" in d:
        return None
    return d["parse"]["wikitext"]

def strip_markup(s):
    s = re.sub(r"\[\[(?:[^\]|]*\|)?([^\]]*)\]\]", r"\1", s)     # [[a|b]] -> b
    s = re.sub(r"\{\{KW\|([^}]*)\}\}", r"\1", s)                 # {{KW|x}} -> x
    s = re.sub(r"\{\{[^}]*\}\}", "", s)                          # outros templates
    s = re.sub(r"'{2,}", "", s)                                  # ''italic'' '''bold'''
    s = re.sub(r"<[^>]+>", "", s)                                # html
    return re.sub(r"\s+", " ", s).strip()

def infobox(wt):
    fields = {}
    for m in re.finditer(r"^\|\s*(\w+)\s*=\s*(.*)$", wt, re.M):
        fields[m.group(1)] = m.group(2).strip()
    return fields

def section(wt, name):
    m = re.search(r"==\s*" + name + r"\s*==\s*\n(.*?)(?=\n==[^=]|\Z)", wt, re.S)
    return m.group(1).strip() if m else ""

names = {q: q for q in QUESTS}
short = {q: q.removesuffix(" Quest") for q in QUESTS}

report = []
for q in QUESTS:
    main = fetch(q); time.sleep(0.5)
    spoiler = fetch(q + "/Spoiler"); time.sleep(0.5)
    (OUT / (q.replace("'", "").replace(" ", "_") + ".json")).write_text(
        json.dumps({"main": main, "spoiler": spoiler}, ensure_ascii=False), encoding="utf-8")
    ib = infobox(main or "")
    report.append(f"\n# {q}\n")
    report.append(f"- lvl: `{ib.get('lvl','')}` | premium: `{ib.get('premium','')}` | aka: `{ib.get('aka','')}`")
    report.append(f"- reward: {strip_markup(ib.get('reward',''))[:300]}")
    report.append(f"- legend: {strip_markup(ib.get('legend',''))[:300]}")
    req = section(spoiler or "", "Requirements")
    report.append("- Requirements:")
    for line in req.splitlines():
        if line.strip():
            report.append(f"    {strip_markup(line)}")
    report.append("- Menções a outras das 15 (fora de Requirements):")
    for src_name, wt in (("main", main), ("spoiler", spoiler)):
        for line in (wt or "").splitlines():
            for other in QUESTS:
                if other == q: continue
                if short[other] in line and "== Requirements ==" not in line:
                    report.append(f"    [{src_name}] ({short[other]}) {strip_markup(line)[:400]}")
                    break
pathlib.Path(__file__).parent.joinpath("report.md").write_text("\n".join(report), encoding="utf-8")
print("ok")
```
